cmake_minimum_required(VERSION 3.24)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../vcpkg.json" TESSERA_MANIFEST)
string(JSON TESSERA_VERSION GET "${TESSERA_MANIFEST}" "version-semver")
string(REGEX REPLACE "-.*$" "" TESSERA_BASE_VERSION "${TESSERA_VERSION}")
project(TesseraNativeUI VERSION "${TESSERA_BASE_VERSION}" LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(TESSERA_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." CACHE PATH "Read-only Tessera source and artwork")
get_filename_component(TESSERA_ROOT "${TESSERA_ROOT}" ABSOLUTE)
if(NOT EXISTS "${TESSERA_ROOT}/include/Match.h")
  message(FATAL_ERROR "Set TESSERA_ROOT to your Tessera source checkout")
endif()
set(IMGUI_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/imgui")
add_library(imgui STATIC ${IMGUI_DIR}/imgui.cpp ${IMGUI_DIR}/imgui_draw.cpp ${IMGUI_DIR}/imgui_tables.cpp ${IMGUI_DIR}/imgui_widgets.cpp)
target_include_directories(imgui PUBLIC ${IMGUI_DIR} ${IMGUI_DIR}/backends)
add_executable(TesseraNativeUI src/GameApp.cpp src/GameInspector.cpp src/GameAlbum.cpp src/GameMatch.cpp src/GameRules.cpp src/GameJournal.cpp src/GameTournaments.cpp src/GameSettings.cpp src/GameShop.cpp src/main.cpp src/App.cpp src/Album.cpp src/Packs.cpp  src/Assets.cpp src/Material.cpp)
set(JSON_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/third_party/nlohmann-json/include")
target_include_directories(TesseraNativeUI PRIVATE src "${TESSERA_ROOT}/include" "${JSON_INCLUDE}")
target_compile_definitions(TesseraNativeUI PRIVATE TESSERA_DEFAULT_ROOT="${TESSERA_ROOT}" TESSERA_PREVIEW_ROOT="${CMAKE_CURRENT_SOURCE_DIR}" IMGUI_DEFINE_MATH_OPERATORS)
target_link_libraries(TesseraNativeUI PRIVATE imgui)
if(WIN32)
  set(WEBP_BUILD_ANIM_UTILS OFF CACHE BOOL "" FORCE)
  set(WEBP_BUILD_CWEBP OFF CACHE BOOL "" FORCE)
  set(WEBP_BUILD_DWEBP OFF CACHE BOOL "" FORCE)
  set(WEBP_BUILD_GIF2WEBP OFF CACHE BOOL "" FORCE)
  set(WEBP_BUILD_IMG2WEBP OFF CACHE BOOL "" FORCE)
  set(WEBP_BUILD_VWEBP OFF CACHE BOOL "" FORCE)
  set(WEBP_BUILD_WEBPINFO OFF CACHE BOOL "" FORCE)
  set(WEBP_BUILD_EXTRAS OFF CACHE BOOL "" FORCE)
  set(WEBP_BUILD_LIBWEBPMUX OFF CACHE BOOL "" FORCE)
  add_subdirectory(third_party/libwebp EXCLUDE_FROM_ALL)
  target_sources(imgui PRIVATE ${IMGUI_DIR}/backends/imgui_impl_win32.cpp ${IMGUI_DIR}/backends/imgui_impl_dx11.cpp)
  target_sources(TesseraNativeUI PRIVATE src/HostDX11.cpp src/RendererDX11.cpp)
  target_compile_definitions(TesseraNativeUI PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(TesseraNativeUI PRIVATE webp d3d11 dxgi d3dcompiler dwmapi imm32)
  add_executable(preview_dx_state_test tests/dx_state_test.cpp src/RendererDX11.cpp src/Material.cpp)
  target_include_directories(preview_dx_state_test PRIVATE src)
  target_compile_definitions(preview_dx_state_test PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(preview_dx_state_test PRIVATE imgui d3d11 dxgi d3dcompiler dwmapi imm32)
  if(MINGW)
    target_link_options(TesseraNativeUI PRIVATE -static -static-libgcc -static-libstdc++)
    target_link_options(preview_dx_state_test PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
  pkg_check_modules(WEBP REQUIRED IMPORTED_TARGET libwebp)
  find_package(OpenGL REQUIRED)
  target_sources(imgui PRIVATE ${IMGUI_DIR}/backends/imgui_impl_sdl2.cpp ${IMGUI_DIR}/backends/imgui_impl_opengl3.cpp)
  target_link_libraries(imgui PUBLIC PkgConfig::SDL2 OpenGL::GL)
  target_sources(TesseraNativeUI PRIVATE src/HostGL.cpp)
  target_link_libraries(TesseraNativeUI PRIVATE PkgConfig::WEBP)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(preview_session_test tests/session_test.cpp)
  target_include_directories(preview_session_test PRIVATE skse ${IMGUI_DIR})
  add_test(NAME preview_session COMMAND preview_session_test)
endif()
if(BUILD_TESTING AND NOT WIN32)
  add_library(native_test_imgui STATIC ${IMGUI_DIR}/imgui.cpp ${IMGUI_DIR}/imgui_draw.cpp ${IMGUI_DIR}/imgui_tables.cpp ${IMGUI_DIR}/imgui_widgets.cpp)
  target_include_directories(native_test_imgui PUBLIC ${IMGUI_DIR})
  target_compile_definitions(native_test_imgui PUBLIC IMGUI_ENABLE_TEST_ENGINE IMGUI_DEFINE_MATH_OPERATORS)
  add_executable(native_game_ui_test tests/game_ui_test.cpp src/App.cpp src/Album.cpp src/Packs.cpp  src/Assets.cpp src/Material.cpp src/GameApp.cpp src/GameInspector.cpp src/GameAlbum.cpp src/GameMatch.cpp src/GameRules.cpp src/GameJournal.cpp src/GameTournaments.cpp src/GameSettings.cpp src/GameShop.cpp)
  target_include_directories(native_game_ui_test PRIVATE src "${TESSERA_ROOT}/include" "${JSON_INCLUDE}")
  target_link_libraries(native_game_ui_test PRIVATE native_test_imgui PkgConfig::WEBP)
  add_test(NAME native_game_ui COMMAND native_game_ui_test "${TESSERA_ROOT}" "${TESSERA_ROOT}/build/test-fixtures")
  set_tests_properties(native_game_ui PROPERTIES FIXTURES_REQUIRED native_fixtures)
endif()
