find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
file(GLOB TESSERA_CORE_TESTS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*_test.cpp")
foreach(source IN LISTS TESSERA_CORE_TESTS)
  get_filename_component(name "${source}" NAME_WE)
  if(name STREQUAL "commonlib_lifetime_test")
    continue()
  endif()
  add_executable(core_${name} "${source}")
  target_include_directories(core_${name} PRIVATE ../include ../native/third_party/nlohmann-json/include)
  target_compile_options(core_${name} PRIVATE -UNDEBUG)
  if(name STREQUAL "completionist_bridge_test")
    target_include_directories(core_${name} BEFORE PRIVATE completionist_stubs)
  endif()
  if(name STREQUAL "drop_in_music_test")
    add_test(NAME ${name} COMMAND core_${name} "${PROJECT_SOURCE_DIR}/Music/TTCG/RPG Medieval Animated Music.xwm")
  else()
    add_test(NAME ${name} COMMAND core_${name})
  endif()
  set_tests_properties(${name} PROPERTIES WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}")
endforeach()
add_test(NAME vendor_distribution_no_pack_injection COMMAND core_vendor_distribution_test --no-pack-injection)
set_tests_properties(vendor_distribution_no_pack_injection PROPERTIES WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}")
foreach(name ui tournament shop)
  add_executable(fixture_${name} ${name}_host.cpp)
  target_include_directories(fixture_${name} PRIVATE ../include ../native/third_party/nlohmann-json/include)
  set_target_properties(fixture_${name} PROPERTIES OUTPUT_NAME "ttcg-${name}-host" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/hosts")
endforeach()
add_test(NAME native_fixtures COMMAND "${Python3_EXECUTABLE}" "${PROJECT_SOURCE_DIR}/tools/native-ui-fixtures.py")
set_tests_properties(native_fixtures PROPERTIES FIXTURES_SETUP native_fixtures ENVIRONMENT "TESSERA_TEST_HOSTS=${CMAKE_BINARY_DIR}/hosts")
file(GLOB TESSERA_PYTHON_TESTS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*_test.py")
foreach(source IN LISTS TESSERA_PYTHON_TESTS)
  get_filename_component(name "${source}" NAME_WE)
  if(name MATCHES "^(native_build_test|release_package_test)$")
    continue()
  endif()
  add_test(NAME python_${name} COMMAND "${Python3_EXECUTABLE}" "${source}")
  set_tests_properties(python_${name} PROPERTIES WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}" ENVIRONMENT "CXX=${CMAKE_CXX_COMPILER};PYTHONUTF8=1")
endforeach()
